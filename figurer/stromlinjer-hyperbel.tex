\begin{tikzpicture}[scale=0.85]
\draw[akse] (-3,0)--(3.2,0) node[right]{$x$};
\draw[akse] (0,-3)--(0,3.2) node[above]{$y$};
% strømlinjer xy = c
\foreach \c in {0.5,1.5,3} {
  \foreach \sx/\sy in {1/1,-1/1,1/-1,-1/-1} {
    \draw[bla,thick,domain={\c/2.9}:2.9,samples=40] plot ({\sx*\x},{\sy*\c/\x});
  }}
% piler på strømlinjene (bevegelse bort fra y-aksen, mot x-aksen)
\foreach \c in {0.5,1.5,3} {
  \foreach \sx/\sy in {1/1,-1/1,1/-1,-1/-1} {
    \pgfmathsetmacro{\xa}{sqrt(\c)}
    \draw[->,bla,very thick] ({\sx*\xa},{\sy*\c/\xa}) -- ({\sx*(\xa+0.05)},{\sy*\c/(\xa+0.05)});
  }}
% feltpiler
\foreach \x in {-2,-1,1,2} \foreach \y in {-2,-1,1,2} \draw[->,black!60] (\x,\y)--++({0.22*\x},{-0.22*\y});
\fill (0,0) circle (1.8pt);
\end{tikzpicture}
